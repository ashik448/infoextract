async function aiStructure(text) { // Keep the original OCR results as backup const backupFields = [...fields]; try { const r = await fetch('/api/extract', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ text, targetLanguage: 'English' }) }); const data = await r.json(); if (!r.ok) { throw new Error(data.error || 'AI unavailable'); } if (!Array.isArray(data.fields) || !data.fields.length) { throw new Error('AI returned no fields'); } const cleanedFields = data.fields .map((f, i) => { const backup = backupFields[i] || {}; return { label: String(f.label || backup.label || 'Information'), original: String(f.original || backup.original || ''), english: String(f.english || ''), confidence: ['high', 'medium', 'low'].includes(f.confidence) ? f.confidence : 'medium' }; }) .filter(f => f.original.trim() || f.english.trim()); if (!cleanedFields.length) { throw new Error('AI returned empty field values'); } fields = cleanedFields; toast('AI organized the information'); $('status').textContent = 'AI extraction completed. Review the results.'; } catch (e) { console.warn('AI extraction error:', e); fields = backupFields; toast('AI unavailable — OCR results preserved'); $('status').textContent = 'AI processing failed. Your original OCR results are preserved. ' + e.message; } }
